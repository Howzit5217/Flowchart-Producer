Start
Declare Integer pick
Display "1. Add   2. Remove   3. List   4. Quit"
Input pick
While pick <> 4
    Select Case pick
        Case 1
            Display "Adding."
        Case 2
            Display "Removing."
        Case 3
            Display "Listing."
        Default
            Display "Not on the menu."
    End Select
    Input pick
End While
End
